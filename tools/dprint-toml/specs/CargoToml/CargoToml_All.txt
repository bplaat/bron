-- Cargo.toml --
== should format "comit" Cargo.toml file ==
[package]
authors = ["Some author"]
name = "package"
description = "Description"
version = "0.1.0"
edition = "2018"

[dependencies]
b = "0.1"
a = { version = "0.8", features = ["serde", "v4"] }

[dev-dependencies]
c = "0.1"
a = { version = "0.8", features = ["serde", "v4"] }
b = "0.2"

[features]
default = []
test = ["feature"]

[expect]
[package]
name = "package"
version = "0.1.0"
authors = ["Some author"]
edition = "2018"
description = "Description"

[features]
default = []
test = ["feature"]

[dependencies]
a = { version = "0.8", features = ["serde", "v4"] }
b = "0.1"

[dev-dependencies]
a = { version = "0.8", features = ["serde", "v4"] }
b = "0.2"
c = "0.1"
